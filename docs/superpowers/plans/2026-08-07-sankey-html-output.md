# Sankey Diagram HTML Output — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the SankeyMATIC text output with a self-contained d3-sankey HTML file that can be hosted on GitHub Pages and embedded in Notion.

**Architecture:** `notion_sankey.py` fetches from Notion and builds flows → calls `render.py` which embeds the flows as JSON into an HTML template with d3-sankey rendering code → writes `output/sankey.html` → git push → GitHub Pages → Notion `/embed`.

**Tech Stack:** Python 3 (requests), d3.js v7 (CDN), d3-sankey v0.12 (CDN), static HTML/CSS/JS. No new Python dependencies.

---

## File Map

| File | Action | Responsibility |
|------|--------|----------------|
| `config.py` | **Create** | User-editable settings: title, colors, layout dimensions |
| `render.py` | **Create** | HTML template + `render_sankey_html()` function |
| `notion_sankey.py` | **Modify** | Replace `print_sankeymatic()` call with `render_sankey_html()` |
| `.gitignore` | **Create** | Ignore `.env` |
| `output/sankey.html` | **Generated** | The output HTML (committed to repo for GitHub Pages) |

```python
"""Generate a standalone Sankey diagram HTML file from flow data."""

import json
from datetime import datetime
from pathlib import Path

import config


# ---------------------------------------------------------------------------
# HTML template — __MARKERS__ are replaced at render time.
# Do not edit this block unless you know what you're doing.
# ---------------------------------------------------------------------------

_TEMPLATE = r"""<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<title>__TITLE__</title>
<style>
  *, *::before, *::after { margin: 0; padding: 0; box-sizing: border-box; }
  body {
    font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, Oxygen,
                 Ubuntu, Cantarell, sans-serif;
    background: #f8f9fa;
    color: #212529;
    display: flex;
    flex-direction: column;
    align-items: center;
    padding: 2rem 1rem;
  }
  header { text-align: center; margin-bottom: 1.5rem; }
  header h1 { font-size: 1.5rem; font-weight: 600; margin-bottom: 0.25rem; }
  header p  { font-size: 0.8rem; color: #6c757d; }
  .container { width: 100%; max-width: __PAGE_WIDTH__px; }
  svg { display: block; margin: 0 auto; }
  .node rect {
    cursor: pointer;
    stroke: #fff;
    stroke-width: 1px;
    shape-rendering: crispEdges;
  }
  .node text {
    font-size: 12px;
    font-weight: 500;
    fill: #333;
    pointer-events: none;
  }
  .link {
    fill: none;
    stroke-opacity: __FLOW_OPACITY__;
    shape-rendering: auto;
  }
  .link:hover { stroke-opacity: 0.7; }
  footer {
    margin-top: 2rem;
    font-size: 0.75rem;
    color: #adb5bd;
  }
</style>
</head>
<body>
<header>
  <h1>__TITLE__</h1>
  <p>Last updated: __UPDATED__</p>
</header>
<div class="container">
  <svg id="sankey"></svg>
</div>
<footer>Generated by notion_sankey.py &middot; Powered by d3.js</footer>

<script src="https://unpkg.com/d3@7/dist/d3.min.js"></script>
<script src="https://unpkg.com/d3-sankey@0.12/dist/d3-sankey.min.js"></script>
<script>
(function () {
  var DATA = __DATA_JSON__;

  var margins = DATA.margins;
  var width  = DATA.pageWidth - margins.left - margins.right;
  var height = DATA.pageHeight - margins.top - margins.bottom;

  // --- Build nodes & links arrays for d3-sankey ---
  var nodeNames = [];
  var seen = {};
  DATA.flows.forEach(function (f) {
    if (!seen[f.source]) { seen[f.source] = true; nodeNames.push(f.source); }
    if (!seen[f.target]) { seen[f.target] = true; nodeNames.push(f.target); }
  });

  var nodeIndex = {};
  nodeNames.forEach(function (n, i) { nodeIndex[n] = i; });

  var nodes = nodeNames.map(function (name) { return { name: name }; });
  var links = DATA.flows.map(function (f) {
    return {
      source: nodeIndex[f.source],
      target: nodeIndex[f.target],
      value:  f.value,
    };
  });

  // --- d3-sankey layout ---
  var sankey = d3.sankey()
    .nodeWidth(DATA.nodeWidth)
    .nodePadding(DATA.nodePadding)
    .extent([[1, 1], [width - 1, height - 1]]);

  var layout = sankey({ nodes: nodes, links: links });
  var layoutNodes = layout.nodes;
  var layoutLinks = layout.links;

  // --- Render ---
  var svg = d3.select("#sankey")
    .attr("width",  width  + margins.left + margins.right)
    .attr("height", height + margins.top  + margins.bottom)
    .append("g")
    .attr("transform", "translate(" + margins.left + "," + margins.top + ")");

  // Color helpers
  function nodeFill(d) {
    return DATA.colors[d.name] || d3.schemeCategory10[nodeNames.indexOf(d.name) % 10];
  }

  function linkStroke(d) {
    return DATA.colors[d.source.name] || "#999";
  }

  // --- Links (drawn first so they sit behind nodes) ---
  svg.append("g")
    .attr("fill", "none")
    .selectAll("path")
    .data(layoutLinks)
    .join("path")
    .attr("class", "link")
    .attr("d", d3.sankeyLinkHorizontal())
    .attr("stroke", linkStroke)
    .attr("stroke-width", function (d) { return Math.max(1, d.width); })
    .append("title")
    .text(function (d) {
      return d.source.name + " → " + d.target.name + ": " + d.value;
    });

  // --- Nodes ---
  var nodeG = svg.append("g")
    .selectAll("g")
    .data(layoutNodes)
    .join("g")
    .attr("class", "node");

  nodeG.append("rect")
    .attr("x", function (d) { return d.x0; })
    .attr("y", function (d) { return d.y0; })
    .attr("width",  function (d) { return d.x1 - d.x0; })
    .attr("height", function (d) { return d.y1 - d.y0; })
    .attr("fill", nodeFill)
    .append("title")
    .text(function (d) { return d.name + ": " + d.value; });

  // --- Labels ---
  nodeG.append("text")
    .attr("x", function (d) {
      return d.x0 < width / 2 ? d.x1 + 6 : d.x0 - 6;
    })
    .attr("y", function (d) { return (d.y0 + d.y1) / 2; })
    .attr("dy", "0.35em")
    .attr("text-anchor", function (d) {
      return d.x0 < width / 2 ? "start" : "end";
    })
    .text(function (d) { return d.name; });
})();
</script>
</body>
</html>"""


# ---------------------------------------------------------------------------
# Public API
# ---------------------------------------------------------------------------

def render_sankey_html(flows, output_path="output/sankey.html"):
    """Generate a standalone sankey.html from a flows dictionary.

    Parameters
    ----------
    flows : dict[tuple[str, str], int]
        Mapping from (source_stage, target_stage) to count.
        E.g. {("Applied", "OA"): 12, ("OA", "Interview"): 8, ...}
    output_path : str
        Path where the HTML file will be written.
        Parent directories are created if they don't exist.
    """
    flows_list = [
        {"source": s, "target": t, "value": v}
        for (s, t), v in sorted(flows.items(), key=lambda kv: -kv[1])
    ]

    now_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    data = {
        "title":   config.TITLE,
        "flows":   flows_list,
        "colors":  config.COLORS,
        "nodeWidth":   config.NODE_WIDTH,
        "nodePadding": config.NODE_PADDING,
        "flowOpacity": config.FLOW_OPACITY,
        "pageWidth":   config.PAGE_WIDTH,
        "pageHeight":  config.PAGE_HEIGHT,
        "margins":     config.MARGINS,
    }

    html = _TEMPLATE
    html = html.replace("__TITLE__",        config.TITLE)
    html = html.replace("__UPDATED__",      now_str)
    html = html.replace("__FLOW_OPACITY__", str(config.FLOW_OPACITY))
    html = html.replace("__PAGE_WIDTH__",   str(config.PAGE_WIDTH))
    html = html.replace("__DATA_JSON__",    json.dumps(data))

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(html, encoding="utf-8")
    print(f"Wrote Sankey diagram to {output_path}")
```
